  const bad=[];
  pfRows.forEach(function(p,i){
    p.name=String(p.name).replace(/\s+/g,' ').trim();p.id=normIdCell(p.id);
    if(!p.name||FORMULA.test(p.name)||!/^\d{5,9}$/.test(p.id))bad.push(i+1);
  });
  if(bad.length){$('pfmsg').innerHTML='<span class="err">שורות '+bad.join(', ')+': חסר שם או שת.ז. לא תקינה (5-9 ספרות). מתקנים בטבלה או מוחקים את השורה, ואז מאשרים שוב.</span>';return}

  const ta=$('people');
  const base=ta.value.replace(/\s+$/,'');
  ta.value=(base?base+'\n':'')+pfRows.map(function(p){return p.name+'\t'+p.id}).join('\n')+'\n';

}
$('peoplefile').onchange=async e=>{
  const f=e.target.files[0];e.target.value='';
  if(!f)return;
  pfWb=null;pfSheets=[];pfSheetName='';pfAOA=null;pfAn=null;pfCols=null;pfRows=[];renderPf();
  $('pfmsg').innerHTML='';
  if(!/\.(xlsx|xls|csv)$/i.test(f.name)){$('pfmsg').innerHTML='<span class="err">נתמכים קבצי ‎.xlsx, ‎.xls, ‎.csv בלבד.</span>';return}
  if(f.size>10e6){$('pfmsg').innerHTML='<span class="err">הקובץ גדול מדי (מעל 10MB).</span>';return}
  try{
    const buf=await f.arrayBuffer();let aoa;
    if(/\.csv$/i.test(f.name)){
      let txt;
      try{txt=new TextDecoder('utf-8',{fatal:true}).decode(buf)}catch(_){txt=new TextDecoder('windows-1255').decode(buf)}
      aoa=parseCSV(txt.replace(/^\uFEFF/,''));pfSheets=[];
    }else{
      pfWb=XLSX.read(buf,{type:'array'});
      pfSheets=pfWb.SheetNames.slice();pfSheetName=pfSheets[0];
      aoa=pfSheetAOA(pfSheetName);
    }
    pfFileName=f.name;pfAOA=aoa;
    const an=analyzeAOA(aoa);
    if(an.err){$('pfmsg').innerHTML='<span class="err">'+esc(an.err)+'</span>';return}
    pfAn=an;pfCols={headerRow:an.headerRow,nameCol:an.nameCol,idCol:an.idCol};
    pfExtract();renderPf();pfMsgReview();
  }catch(err){
    pfWb=null;pfSheets=[];pfSheetName='';pfAOA=null;pfAn=null;pfCols=null;pfRows=[];renderPf();
    $('pfmsg').innerHTML='<span class="err">לא הצלחתי לקרוא את הקובץ. נתמכים ‎.xlsx, ‎.xls, ‎.csv.</span>';
  }
};
